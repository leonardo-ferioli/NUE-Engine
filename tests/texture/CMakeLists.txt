# Texture module tests (console only). Picked up automatically: tests/texture/test_<name>.cpp -> ctest name nue_texture_test_<name>.
file(GLOB NUE_TEXTURE_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/test_*.cpp")
foreach(src IN LISTS NUE_TEXTURE_TEST_SOURCES)
    get_filename_component(name "${src}" NAME_WE)
    string(REGEX REPLACE "^test_" "" short "${name}")
    set(target nue_texture_test_${short})
    add_executable(${target} "${src}")
    target_link_libraries(${target} PRIVATE ${NUE_TEXTURE_TEST_LIB})
    if(NUE_TEXTURE_STANDALONE)
        target_compile_definitions(${target} PRIVATE NUE_TEXTURE_TEST_INTERNALS=1)
    else()
        target_compile_definitions(${target} PRIVATE NUE_TEXTURE_WITH_ENGINE=1)
    endif()
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/.." "${NUE_ROOT}/src/texture")
    if(WIN32)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    endif()
    if(short STREQUAL "simd")
        target_compile_definitions(${target} PRIVATE NUE_TEX_SIMD_GOLDEN_PATH="${CMAKE_CURRENT_LIST_DIR}/data/simd_golden.bin")
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 300)
endforeach()
