# NUE texture module: library nue_texture (+ tests/texture, tools/nue_texconv).
#
# Two ways to build it:
#   * as a part of the engine:  the root CMakeLists calls add_subdirectory(src/texture)  (request in docs/REQUESTS_texture.md)
#   * standalone (no SDL, no engine):  cmake -S src/texture -B build_tex -G Ninja -DCMAKE_C_COMPILER=clang -DCMAKE_CXX_COMPILER=clang++
#                                      cmake --build build_tex && ctest --test-dir build_tex
#
# Options
#   NUE_TEXTURE_SIMD     auto | none | sse2 | sse41 | avx2 | neon     (compile-time ISA of astc-encoder and etcpak; auto = sse41 on x86, neon on arm64)
#   NUE_TEXTURE_TESTS    build tests/texture (default ON when top level)
#   NUE_TEXTURE_TOOLS    build tools/nue_texconv (default ON when top level)
cmake_minimum_required(VERSION 3.21)

if(NOT PROJECT_NAME)
    project(nue_texture VERSION 0.1.0 LANGUAGES C CXX)
    set(NUE_TEXTURE_TOP_LEVEL ON)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
    endif()
    enable_testing()
else()
    set(NUE_TEXTURE_TOP_LEVEL OFF)
endif()

set(NUE_TEXTURE_SIMD "auto" CACHE STRING "SIMD level of the codecs: auto|none|sse2|sse41|avx2|neon")
if(NUE_TEXTURE_TOP_LEVEL)
    set(_default_tests ON)
else()
    set(_default_tests ${NUE_BUILD_TESTS})
endif()
option(NUE_TEXTURE_TESTS "Build the texture tests" ${_default_tests})
option(NUE_TEXTURE_TOOLS "Build nue_texconv" ${_default_tests})

get_filename_component(NUE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(TP "${CMAKE_CURRENT_LIST_DIR}/third_party")

# ---------------------------------------------------------------------------------------------------- ISA
set(_simd "${NUE_TEXTURE_SIMD}")
if(_simd STREQUAL "auto")
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i[3-6]86|x86)$" OR CMAKE_CXX_COMPILER_TARGET MATCHES "x86_64|i686")
        set(_simd sse41)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$" OR CMAKE_CXX_COMPILER_TARGET MATCHES "aarch64|arm64")
        set(_simd neon)
    else()
        set(_simd none)
    endif()
endif()
message(STATUS "NUE texture: codec SIMD level ${_simd}")

if(MSVC AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    set(_gnu_fe OFF)
else()
    set(_gnu_fe ON)
endif()

set(_astc_defs ASTCENC_NO_INVARIANCE=1)
set(_isa_opts "")
if(_simd STREQUAL "none")
    list(APPEND _astc_defs ASTCENC_NEON=0 ASTCENC_SVE=0 ASTCENC_SSE=0 ASTCENC_AVX=0 ASTCENC_POPCNT=0 ASTCENC_F16C=0)
elseif(_simd STREQUAL "neon")
    list(APPEND _astc_defs ASTCENC_NEON=1 ASTCENC_SVE=0 ASTCENC_SSE=0 ASTCENC_AVX=0 ASTCENC_POPCNT=0 ASTCENC_F16C=0)
elseif(_simd STREQUAL "sse2")
    list(APPEND _astc_defs ASTCENC_NEON=0 ASTCENC_SVE=0 ASTCENC_SSE=20 ASTCENC_AVX=0 ASTCENC_POPCNT=0 ASTCENC_F16C=0 ASTCENC_X86_GATHERS=0)
    if(_gnu_fe)
        set(_isa_opts -msse2)
    endif()
elseif(_simd STREQUAL "sse41")
    list(APPEND _astc_defs ASTCENC_NEON=0 ASTCENC_SVE=0 ASTCENC_SSE=41 ASTCENC_AVX=0 ASTCENC_POPCNT=1 ASTCENC_F16C=0 ASTCENC_X86_GATHERS=0)
    if(_gnu_fe)
        set(_isa_opts -msse4.1 -mpopcnt)
    endif()
elseif(_simd STREQUAL "avx2")
    list(APPEND _astc_defs ASTCENC_NEON=0 ASTCENC_SVE=0 ASTCENC_SSE=41 ASTCENC_AVX=2 ASTCENC_POPCNT=1 ASTCENC_F16C=1 ASTCENC_X86_GATHERS=0)
    if(_gnu_fe)
        set(_isa_opts -mavx2 -mfma -mf16c -mpopcnt -mbmi2)
    else()
        set(_isa_opts /arch:AVX2)
    endif()
else()
    message(FATAL_ERROR "NUE_TEXTURE_SIMD: unknown value ${_simd}")
endif()

# ---------------------------------------------------------------------------------------------------- vendored codecs (each its own static library)
file(GLOB _astc_src CONFIGURE_DEPENDS "${TP}/astc-encoder/astcenc_*.cpp")
add_library(nue_tex_astc STATIC ${_astc_src})
target_include_directories(nue_tex_astc PUBLIC "${TP}/astc-encoder")
target_compile_definitions(nue_tex_astc PRIVATE ${_astc_defs} _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(nue_tex_astc PRIVATE cxx_std_17)
target_compile_options(nue_tex_astc PRIVATE ${_isa_opts})
if(_gnu_fe)
    target_compile_options(nue_tex_astc PRIVATE -ffp-contract=off -fno-math-errno -w)
endif()

set(_etc_src ProcessRGB.cpp ProcessDxtc.cpp Decode.cpp Tables.cpp Dither.cpp bc7enc.cpp)
list(TRANSFORM _etc_src PREPEND "${TP}/etcpak/")
add_library(nue_tex_etcpak STATIC ${_etc_src})
target_include_directories(nue_tex_etcpak PUBLIC "${TP}/etcpak" PRIVATE "${TP}/bcdec")
target_compile_features(nue_tex_etcpak PRIVATE cxx_std_20)   # bc7enc.cpp uses <bit>
target_compile_definitions(nue_tex_etcpak PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
if(_gnu_fe)
    target_compile_options(nue_tex_etcpak PRIVATE ${_isa_opts} -w)
elseif(_simd STREQUAL "avx2")
    target_compile_options(nue_tex_etcpak PRIVATE ${_isa_opts})
endif()
if(_simd STREQUAL "avx2")
    target_compile_definitions(nue_tex_etcpak PRIVATE __AVX2__=1)
endif()

set(_pvr_src BitScale.cpp MortonTable.cpp PvrTcDecoder.cpp PvrTcEncoder.cpp PvrTcPacket.cpp)
list(TRANSFORM _pvr_src PREPEND "${TP}/pvrtc_javelin/")
add_library(nue_tex_pvrtc STATIC ${_pvr_src})
target_include_directories(nue_tex_pvrtc PUBLIC "${TP}/pvrtc_javelin")
target_compile_features(nue_tex_pvrtc PRIVATE cxx_std_17)
target_compile_definitions(nue_tex_pvrtc PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(_gnu_fe)
    target_compile_options(nue_tex_pvrtc PRIVATE -w)
endif()

# ---------------------------------------------------------------------------------------------------- the module
file(GLOB NUE_TEXTURE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/*.cpp")
if(TARGET nue_texture_obj)
    # Part of the engine (root CMakeLists contract): the sources go into nue_texture_obj, the vendored codecs are forwarded to libnue.
    target_sources(nue_texture_obj PRIVATE ${NUE_TEXTURE_SOURCES})
    target_include_directories(nue_texture_obj PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${TP}/bcdec" "${TP}/stb")
    target_link_libraries(nue_texture_obj PUBLIC nue_tex_astc nue_tex_etcpak nue_tex_pvrtc)
    target_compile_definitions(nue_texture_obj PRIVATE NUE_TEX_SIMD_NAME="${_simd}")
    set(_nue_texture_target nue_texture_obj)
    set(NUE_TEXTURE_TEST_LIB nue::nue)
else()
    add_library(nue_texture STATIC ${NUE_TEXTURE_SOURCES})
    add_library(nue::texture ALIAS nue_texture)
    target_include_directories(nue_texture PUBLIC "${NUE_ROOT}/include" PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${TP}/bcdec" "${TP}/stb")
    target_compile_features(nue_texture PUBLIC cxx_std_17)
    target_link_libraries(nue_texture PRIVATE nue_tex_astc nue_tex_etcpak nue_tex_pvrtc)
    target_compile_definitions(nue_texture PRIVATE NUE_TEX_SIMD_NAME="${_simd}")
    if(NOT WIN32)
        find_package(Threads REQUIRED)
        target_link_libraries(nue_texture PUBLIC Threads::Threads)
    endif()
    set(_nue_texture_target nue_texture)
    set(NUE_TEXTURE_TEST_LIB nue_texture)
    set(NUE_TEXTURE_STANDALONE ON)
endif()
if(MSVC AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(${_nue_texture_target} PRIVATE /W4 /utf-8)
    target_compile_definitions(${_nue_texture_target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
    target_compile_options(${_nue_texture_target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
    if(WIN32)
        target_compile_definitions(${_nue_texture_target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    endif()
endif()

# ---------------------------------------------------------------------------------------------------- tests and tools
if(NUE_TEXTURE_TESTS AND EXISTS "${NUE_ROOT}/tests/texture/CMakeLists.txt")
    add_subdirectory("${NUE_ROOT}/tests/texture" "${CMAKE_CURRENT_BINARY_DIR}/tests_texture")
endif()
if(NUE_TEXTURE_TOOLS AND EXISTS "${NUE_ROOT}/tools/nue_texconv/CMakeLists.txt")
    add_subdirectory("${NUE_ROOT}/tools/nue_texconv" "${CMAKE_CURRENT_BINARY_DIR}/nue_texconv")
endif()
